-- Session table (প্রতিটি recording এর summary)
CREATE TABLE ecg_sessions (
    id            BIGSERIAL PRIMARY KEY,
    name          TEXT NOT NULL,
    started_at    TIMESTAMPTZ NOT NULL,
    ended_at      TIMESTAMPTZ NOT NULL,
    sample_count  INTEGER NOT NULL DEFAULT 0,
    avg_bpm       NUMERIC(6,1) NOT NULL DEFAULT 0,
    created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- Sample table (প্রতিটি row এর raw data)
CREATE TABLE ecg_session_samples (
    id            BIGSERIAL PRIMARY KEY,
    session_id    BIGINT NOT NULL REFERENCES ecg_sessions(id) ON DELETE CASCADE,
    created_at    TIMESTAMPTZ NOT NULL,
    raw_ecg       NUMERIC(10,4) NOT NULL,
    filtered_ecg  NUMERIC(14,4) NOT NULL,
    bpm           SMALLINT NOT NULL DEFAULT 0,
    finger        SMALLINT NOT NULL DEFAULT 0
);

-- Index for fast lookup by session
CREATE INDEX idx_samples_session_id ON ecg_session_samples(session_id);
CREATE INDEX idx_samples_created_at ON ecg_session_samples(created_at);

-- RLS (Row Level Security) — anon key দিয়ে insert/select করতে পারবে
ALTER TABLE ecg_sessions       ENABLE ROW LEVEL SECURITY;
ALTER TABLE ecg_session_samples ENABLE ROW LEVEL SECURITY;

CREATE POLICY "allow_all_ecg_sessions"
    ON ecg_sessions FOR ALL TO anon USING (true) WITH CHECK (true);

CREATE POLICY "allow_all_ecg_samples"
    ON ecg_session_samples FOR ALL TO anon USING (true) WITH CHECK (true);




    -- Storage access policy
CREATE POLICY "public read ecg"
ON storage.objects FOR SELECT TO anon
USING (bucket_id = 'ecg-recordings');

CREATE POLICY "public upload ecg"
ON storage.objects FOR INSERT TO anon
WITH CHECK (bucket_id = 'ecg-recordings');


CREATE POLICY "allow anon delete"
ON storage.objects FOR DELETE TO anon
USING (bucket_id = 'ecg-recordings');
